## 3. Robust detection
1. Probe a fixed list of absolute paths with `fs.access(X_OK)`: Toolbox scripts dir, /usr/local/bin, /opt/homebrew/bin, app-bundle CLIs under /Applications and ~/Applications. (Which of these exist on this machine: see section 1.)
2. Prefer bundle paths over PATH symlinks (symlinks can be dangling after app moves).
3. For WebStorm prefer `/usr/bin/open -na <bundle>` (no dependence on Toolbox script, avoids its literal-quote wrapping of args with spaces).
4. Optional PATH repair: `fix-path` npm package or one `$SHELL -ilc 'echo -n $PATH'` call with timeout [UNVERIFIED].
5. Non-standard installs: `mdfind "kMDItemCFBundleIdentifier == 'com.jetbrains.WebStorm'"` [UNVERIFIED; bundle ids from memory].
6. Cache the result; re-detect lazily after a failed spawn.

## 4. Safe spawning
- `spawn(cmd, argsArray, {shell:false, detached:true, stdio:'ignore'})` + `unref()`; wait for `spawn` or `error` event (ENOENT) to fall through to next editor. Weird names were passed safely in the test above [V-LOCAL].
- Always pass absolute paths (start with `/`, so never parsed as an option).
- Only call `shell.openExternal` with URL schemes you build yourself.

## 5. Playwright + Electron: proven on this machine

- With DevTools open, `app.windows()` listed two pages (the app and a `devtools://` page); in this run `firstWindow()` still returned the app page, so the reported bug did not reproduce [V-LOCAL]. The reported bug (firstWindow returns DevTools in electron-vite dev) is [UNVERIFIED] (search snippet; source page 429). ## 6. Snippet: editor launcher (src/main/editors.ts) [UNVERIFIED as a module; the commands and args are what section 2 ran]
```ts
import { spawn } from 'node:child_process'
import { access, constants } from 'node:fs/promises'
import { homedir } from 'node:os'
import { join } from 'node:path'
import { shell } from 'electron'

type Cmd = { bin: string; args: string[] }
type Editor = { id: string; name: string; candidates: string[]; build: (bin: string, f: string, l: number, c: number) => Cmd }
const H = homedir()
const app = (n: string) => ['/Applications', join(H, 'Applications')].map((d) => `${d}/${n}.app`)

export const EDITORS: Editor[] = [
  { id: 'webstorm', name: 'WebStorm', candidates: app('WebStorm'),
    // candidate is the .app bundle; launched through /usr/bin/open (same as the Toolbox script does)
    build: (bundle, f, l, c) => ({ bin: '/usr/bin/open', args: ['-na', bundle, '--args', '--line', String(l), '--column', String(c), f] }) },
  { id: 'cursor', name: 'Cursor',
    candidates: [...app('Cursor').map((a) => `${a}/Contents/Resources/app/bin/cursor`), '/usr/local/bin/cursor', '/opt/homebrew/bin/cursor'],
    build: (bin, f, l, c) => ({ bin, args: ['-g', `${f}:${l}:${c}`] }) },
  { id: 'vscode', name: 'VS Code',
    candidates: [...app('Visual Studio Code').map((a) => `${a}/Contents/Resources/app/bin/code`), '/usr/local/bin/code', '/opt/homebrew/bin/code'],
    build: (bin, f, l, c) => ({ bin, args: ['-g', `${f}:${l}:${c}`] }) },
  { id: 'zed', name: 'Zed', // UNVERIFIED
    candidates: ['/Applications/Zed.app/Contents/MacOS/cli', '/usr/local/bin/zed', '/opt/homebrew/bin/zed'],
    build: (bin, f, l, c) => ({ bin, args: [`${f}:${l}:${c}`] }) },
  { id: 'sublime', name: 'Sublime Text', // UNVERIFIED
    candidates: ['/Applications/Sublime Text.app/Contents/SharedSupport/bin/subl', '/usr/local/bin/subl', '/opt/homebrew/bin/subl'],
    build: (bin, f, l, c) => ({ bin, args: [`${f}:${l}:${c}`] }) },
]

// directories pass X_OK, so one check covers both .app bundles and binaries
const ok = (p: string) => access(p, constants.X_OK).then(() => true, () => false)
let cache: { editor: Editor; path: string }[] | null = null
export async function detectEditors(force = false) {
  if (cache && !force) return cache
  const found: { editor: Editor; path: string }[] = []
  for (const editor of EDITORS)
    for (const p of editor.candidates) if (await ok(p)) { found.push({ editor, path: p }); break }
  return (cache = found)
}

function run({ bin, args }: Cmd): Promise<boolean> {
  return new Promise((resolve) => {
    const child = spawn(bin, args, { shell: false, detached: true, stdio: 'ignore' })
    child.once('error', () => resolve(false))
    child.once('spawn', () => { child.unref(); resolve(true) })
  })
}

/** file must be absolute. preferred = editor id from settings. Returns editor id used. */
export async function openInEditor(file: string, line = 1, col = 1, preferred?: string) {
  const list = [...(await detectEditors())].sort((a, b) => Number(b.editor.id === preferred) - Number(a.editor.id === preferred))
  for (const { editor, path } of list) if (await run(editor.build(path, file, line, col))) return editor.id
  return (await shell.openPath(file)) === '' ? 'system' : null
}
```

## 7. Snippet: Playwright helper (e2e/helpers.ts) [launch/window/screenshot logic mirrors the proven smoke test; makeRepo UNVERIFIED]
```ts
import { _electron as electron, ElectronApplication, Page } from '@playwright/test'
import { mkdtempSync, mkdirSync, writeFileSync, existsSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join, resolve } from 'node:path'

export async function launch(extraEnv: Record<string, string> = {}) {
  const userData = mkdtempSync(join(tmpdir(), 'fugly-ud-'))
  const env = { ...process.env, E2E: '1', ...extraEnv } as Record<string, string>
  delete env.ELECTRON_RUN_AS_NODE
  const app: ElectronApplication = await electron.launch({
    args: [resolve('.'), `--user-data-dir=${userData}`], // run `electron-vite build` first; package.json main -> out/main/index.js
    cwd: resolve('.'), env, timeout: 30_000,
  })
  await app.firstWindow() // waits for any window
  let page: Page | undefined
  for (let i = 0; i < 50 && !page; i++) {
    page = app.windows().find((p) => !p.url().startsWith('devtools://'))
    if (!page) await new Promise((r) => setTimeout(r, 100))
  }
  if (!page) throw new Error('main window not found')
  page.on('console', (m) => console.log('[renderer]', m.text()))
  app.process().stderr?.on('data', (d) => process.stderr.write(`[main] ${d}`))
  await page.waitForLoadState('domcontentloaded')
  return { app, page, userData }
}
export const shot = (page: Page, name: string) => page.screenshot({ path: `e2e/screenshots/${name}.png` })

/** ~files*linesPer lines (default 500k). Cached by marker file. */
export function makeRepo(root = join(tmpdir(), 'fugly-500k'), files = 5000, linesPer = 100) {
  const marker = join(root, '.done')
  if (existsSync(marker)) return root
  const w = ['alpha', 'beta', 'gamma', 'delta', 'render', 'search', 'index', 'token', 'parse', 'fuzzy']
  for (let i = 0; i < files; i++) {
    const dir = join(root, `d${i % 50}`, `s${i % 7}`)
    mkdirSync(dir, { recursive: true })
    const lines: string[] = []
    for (let l = 0; l < linesPer; l++) lines.push(`const ${w[(i + l) % 10]}_${i}_${l} = ${w[(i * l) % 10]}(${l}) // line ${l}`)
    writeFileSync(join(dir, `file${i}.ts`), lines.join('\n'))
  }
  writeFileSync(marker, '')
  return root
}
```
